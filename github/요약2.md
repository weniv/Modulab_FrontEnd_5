# 브랜치 전략을 사용한 협업

* Feature branch(main, develop, feature) 전략을 사용해보겠습니다.

```
# 이번에는 3개의 브랜치를 만들어서
# header, body-main, footer로 나누어 작업을 해보겠습니다.

# 새로운 폴더를 만들고 이동합니다.
# VSCode로 해당 폴더 열기
# 터미널 열어 아래 명령어 실행

# git 저장소 초기화
git init

# 브랜치 확인(아무것도 없음)
git branch

# index.html 파일을 생성
# 일부러 충돌이 나게 하는 학습도 해보세요.
echo ''>index.html

# 이 명령어 직전에 index.html 파일을 미리 작성해두세요.
git add .
git commit -m "1"

# Feature branch
# main, dev
git branch dev

# feat 브랜치 생성
git branch header
git branch body-main
git branch footer

# 브랜치가 잘 만들어졌는지 확인합니다.
# 지금은 main 브랜치에 있습니다.
git branch

# header 브랜치로 이동하여 작업해주세요.
git checkout header
echo . > header.js
git add .
git commit -m "header"


# body-main 브랜치로 이동하여 작업합니다.
git checkout body-main
echo . > body-main.js
git add .
git commit -m "body-main"

# footer 작업합니다.
git checkout footer
echo . > footer.js
git add .
git commit -m "footer"

# 합칠 때에는 꼭 dev에 들어와 있는지 확인해야 합니다.
# 합칠 때 뜨는 에디터가 있을 텐데 메시지를 다 지워주시고 test-1, test-2, test-3 등으로 작성해주세요.
# 기본 VSCode 사용하도록 글로벌 설정 명령어
# git config --global core.editor "code --wait"
git checkout dev
git merge header
git merge body-main
git merge footer

# main 브랜치로 이동합니다.
git checkout main
git merge dev

# 다른 브렌치 삭제
git branch -d header
git branch -d body-main
git branch -d footer
```
